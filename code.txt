/* ---------- Loading PDFs ----------
   The file itself is NOT read into memory here. Small files are handed to pdf.js whole;
   big files are read by pdf.js in pieces (range requests on the File), so a 1–2 GB PDF
   opens quickly and only the pages you look at are read. The full bytes are read only
   when you download, one source file at a time. */
function isPdfFile(f){ return /\.pdf$/i.test(f.name) || f.type === 'application/pdf'; }
function isImageFile(f){ return (/^image\//.test(f.type) && !/svg/.test(f.type)) || /\.(jpe?g|jfif|png|webp|gif|bmp|avif|heic|heif|tiff?|ico)$/i.test(f.name); }
async function openWithPdfJs(file, head){
  var common = { isEvalSupported:false, useSystemFonts:true };
  if (file.size <= BIG_FILE) {
    var all = new Uint8Array(await file.arrayBuffer());
    return pdfjsLib.getDocument(Object.assign({ data:all }, common)).promise;
  }
  var tr = new pdfjsLib.PDFDataRangeTransport(file.size, head);
  tr.requestDataRange = function(begin, end){
    file.slice(begin, end).arrayBuffer().then(function(buf){ tr.onDataRange(begin, new Uint8Array(buf)); }, function(){});
  };
  return pdfjsLib.getDocument(Object.assign({ range:tr, length:file.size, rangeChunkSize:1 << 20, disableAutoFetch:true, disableStream:true }, common)).promise;
}
async function loadSource(file){
  await engineReady;
  if (!isPdfFile(file)) throw new Error('"' + file.name + '" is not a PDF file.');
  if (!file.size) throw new Error('"' + file.name + '" is empty.');
  var head = new Uint8Array(await file.slice(0, Math.min(file.size, 1 << 20)).arrayBuffer());
  var hs = ''; for (var i=0;i<Math.min(1024, head.length);i++) hs += String.fromCharCode(head[i]);
  if (hs.indexOf('%PDF-') === -1) throw new Error('"' + file.name + '" is not a valid PDF (the PDF header is missing).');
  var doc;
  try {
    doc = await openWithPdfJs(file, head);
  } catch(e) {
    if (e && e.name === 'PasswordException') throw new Error('"' + file.name + '" is password-protected. Remove the password in your PDF app, then upload it again.');
    throw new Error('"' + file.name + '" couldn\u2019t be opened. The file may be damaged or incomplete.');
  }
  var encrypted = false;
  try { var md = await doc.getMetadata(); encrypted = !!(md && md.info && md.info.EncryptFilterName); } catch(e){}
  if (encrypted) { try { doc.destroy(); } catch(e){} throw new Error('"' + file.name + '" is encrypted with security restrictions. Remove the document security, then upload it again.'); }
  if (!doc.numPages) throw new Error('"' + file.name + '" has no pages.');
  var id = 's' + (S.src++);
  var src = { id:id, name:file.name, size:file.size, file:file, doc:doc, count:doc.numPages, color:COLORS[(S.order.length) % COLORS.length], pageCache:new Map() };
  S.sources.set(id, src); S.order.push(id);
  return src;
}
async function ingest(files){
  files = Array.from(files || []);
  if (!files.length) return [];
  var out = [], errors = [];
  progress.show(files.length > 1 ? 'Opening ' + files.length + ' PDFs' : 'Opening ' + files[0].name, 0.05);
  for (var i=0;i<files.length;i++) {
    progress.set('Opening ' + files[i].name + (files.length > 1 ? ' (' + (i+1) + ' of ' + files.length + ')' : ''), (i + .5) / files.length);
    try { out.push(await loadSource(files[i])); } catch(e) { errors.push(e.message || String(e)); }
    await tick();
  }
  progress.hide();
  errors.forEach(function(m){ toast(m, 'error'); });
  renderFiles();
  return out;
}

var IMAGE_ACCEPT = 'image/jpeg,image/png,image/webp,image/gif,image/bmp,image/avif,image/heic,image/tiff,.jpg,.jpeg,.jfif,.png,.webp,.gif,.bmp,.avif,.heic,.tif,.tiff';

/* ---------- Loading images ----------
   JPEG: embedded byte-for-byte (no re-encode). PNG: embedded losslessly.
   Other formats (WEBP, GIF, BMP, AVIF…): decoded once at full size and stored as lossless PNG. */
function jpegOrientation(b){
  if (b[0] !== 0xFF || b[1] !== 0xD8) return 1;
  var i = 2;
  while (i + 9 < b.length) {
    if (b[i] !== 0xFF) return 1;
    var mk = b[i+1], len = (b[i+2] << 8) | b[i+3];
    if (mk === 0xE1 && b[i+4] === 0x45 && b[i+5] === 0x78 && b[i+6] === 0x69 && b[i+7] === 0x66) {
      var t = i + 10, le = b[t] === 0x49;
      var r16 = function(o){ return le ? (b[o] | (b[o+1] << 8)) : ((b[o] << 8) | b[o+1]); };
      var r32 = function(o){ return le ? ((b[o] | (b[o+1] << 8) | (b[o+2] << 16) | (b[o+3] << 24)) >>> 0) : (((b[o] << 24) | (b[o+1] << 16) | (b[o+2] << 8) | b[o+3]) >>> 0); };
      var ifd = t + r32(t + 4); if (ifd + 2 > b.length) return 1;
      var n = r16(ifd);
      for (var k=0;k<n;k++) { var e = ifd + 2 + k * 12; if (e + 10 > b.length) break; if (r16(e) === 0x0112) return r16(e + 8) || 1; }
      return 1;
    }
    if (mk === 0xDA) return 1;
    i += 2 + len;
  }
  return 1;
}
async function loadImage(file){
  if (/svg/i.test(file.type) || /\.svg$/i.test(file.name)) throw new Error('"' + file.name + '" is an SVG. Save it as PDF or PNG first.');
  if (!file.size) throw new Error('"' + file.name + '" is empty.');
  var bytes = new Uint8Array(await file.arrayBuffer());
  var url = URL.createObjectURL(file);
  var el = new Image(); el.decoding = 'async'; el.src = url;
  try { await el.decode(); } catch(e) { URL.revokeObjectURL(url); throw new Error('"' + file.name + '" couldn\u2019t be read as an image. This browser may not support its format.'); }
  var w = el.naturalWidth, h = el.naturalHeight;
  if (!w || !h) { URL.revokeObjectURL(url); throw new Error('"' + file.name + '" has no readable size.'); }
  var isJpg = bytes[0] === 0xFF && bytes[1] === 0xD8 && bytes[2] === 0xFF;
  var isPng = bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4E && bytes[3] === 0x47;
  var fmt = null, data = bytes, orient = 1, converted = false;
  if (isJpg) { orient = jpegOrientation(bytes); if ([1,3,6,8].indexOf(orient) >= 0) fmt = 'jpg'; }
  else if (isPng) fmt = 'png';
  if (!fmt) {
    if (w * h > 120e6 || w > 16384 || h > 16384) { URL.revokeObjectURL(url); throw new Error('"' + file.name + '" is too large to convert here. Save it as JPG or PNG first.'); }
    var c = document.createElement('canvas'); c.width = w; c.height = h;
    c.getContext('2d').drawImage(el, 0, 0);
    var blob = await new Promise(function(r){ c.toBlob(r, 'image/png'); });
    c.width = c.height = 0;
    if (!blob) { URL.revokeObjectURL(url); throw new Error('"' + file.name + '" couldn\u2019t be converted.'); }
    data = new Uint8Array(await blob.arrayBuffer()); fmt = 'png'; orient = 1; converted = true;
  }
  var id = 'i' + (S.img++);
  var im = { id:id, name:file.name, size:file.size, bytes:data, fmt:fmt, w:w, h:h, orient:orient, url:url, el:el, converted:converted };
  S.images.set(id, im);
  return im;
}

/* ---------- Page geometry ---------- */
function fitBox(pw, ph, iw, ih, m){
  var aw = Math.max(1, pw - 2 * m), ah = Math.max(1, ph - 2 * m);
  var s = Math.min(aw / iw, ah / ih), w = iw * s, h = ih * s;
  return { x:(pw - w) / 2, y:(ph - h) / 2, w:w, h:h };
}
function orientSize(base, land){ var s = Math.min(base[0], base[1]), l = Math.max(base[0], base[1]); return land ? [l, s] : [s, l]; }
function imagePageSize(im, o, ref){
  var m = o.margin || 0;
  if (o.size === 'image') {
    var pw = im.w * 0.75, ph = im.h * 0.75; // 96 dpi
    var k = Math.min(1, (14400 - 2 * m) / Math.max(pw, ph));
    return [pw * k + 2 * m, ph * k + 2 * m];
  }
  var base = o.size === 'a4' ? A4 : o.size === 'letter' ? LETTER : ref;
  var land = o.orient === 'landscape' || (o.orient === 'auto' && im.w > im.h);
  return orientSize(base, land);
}
async function pageSizeOf(p){

  return [v.width, v.height];
}

function rotateCanvas(c, rot){
  if (!rot) return c;
  var r = document.createElement('canvas'), sw = rot % 180 !== 0;
  r.width = sw ? c.height : c.width; r.height = sw ? c.width : c.height;

  c.width = c.height = 0;
  return r;
}
function drawSynthetic(p, longSide){
  var s = longSide / Math.max(p.w, p.h);
  var c = document.createElement('canvas'); c.width = Math.max(1, Math.round(p.w * s)); c.height = Math.max(1, Math.round(p.h * s));

  if (p.kind === 'img') {
    var im = S.images.get(p.img), b = fitBox(p.w, p.h, im.w, im.h, p.m);
    drawImageHQ(ctx, im.el, b.x * s, (p.h - b.y - b.h) * s, b.w * s, b.h * s);
  }
  return rotateCanvas(c, p.rot);

var thumbs = new Map(), tq = [], tRunning = 0;
function getThumb(p){
  var key = tkey(p), t = thumbs.get(key);
  if (t) return t.promise;
  t = {}; t.promise = new Promise(function(res, rej){ t.res = res; t.rej = rej; });
  thumbs.set(key, t);
  tq.push({ key:key, p:Object.assign({}, p), t:t });
  pump();
  return t.promise;
}
function pump(){
  while (tRunning < 2 && tq.length) {
    var job = tq.pop(); tRunning++;
    (function(job){
      renderThumb(job).then(function(url){ job.t.res(url); }, function(err){ thumbs.delete(job.key); job.t.rej(err); })
        .then(function(){ tRunning--; pump(); });
    })(job);
  }
}
async function renderThumb(job){
  var p = job.p, target = 320 * DPR(), c;
  if (p.kind !== 'pdf') {
    c = drawSynthetic(p, target);
  } else {
    var page = await getPdfPage(S.sources.get(p.src), p.idx);
    var rotation = (page.rotate + p.rot) % 360;
    var v1 = page.getViewport({ scale:1, rotation:rotation });
    var vp = page.getViewport({ scale: target / Math.max(v1.width, v1.height), rotation:rotation });
    c = document.createElement('canvas'); c.width = Math.ceil(vp.width); c.height = Math.ceil(vp.height);
    var ctx = c.getContext('2d'); ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, c.width, c.height);
    await page.render({ canvasContext:ctx, viewport:vp }).promise;
  }
  var blob = await new Promise(function(r){ c.toBlob(r, 'image/webp', 0.97); });
  c.width = c.height = 0;
  if (!blob) throw new Error('thumb');
  return URL.createObjectURL(blob);
}

/* ---------- Page-number lists: "1-19", "1 to 19", "3, 7, 10-12" ---------- */
function parsePageList(str, n){
  var set = new Set(), parts = String(str || '').split(/[,;]+/);
  for (var i=0;i<parts.length;i++) {
    var t = parts[i].trim().toLowerCase();
    if (!t) continue;
    var m = t.match(/^(\d+)\s*(?:-|\u2013|\u2014|to|se|till|until)\s*(\d+)$/);
    if (m) {
      var a = +m[1], b = +m[2]; if (a > b) { var x = a; a = b; b = x; }
      if (a < 1 || b > n) return { err:'Page ' + (a < 1 ? a : b) + ' doesn\u2019t exist. This document has ' + plural(n, 'page') + '.' };
      for (var k=a;k<=b;k++) set.add(k - 1);
    } else if (/^\d+(\s+\d+)*$/.test(t)) {
      var nums = t.split(/\s+/);
      for (var j=0;j<nums.length;j++) {
        var v = +nums[j];
        if (v < 1 || v > n) return { err:'Page ' + v + ' doesn\u2019t exist. This document has ' + plural(n, 'page') + '.' };
        set.add(v - 1);
      }
    } else {
      return { err:'Couldn\u2019t read \u201c' + parts[i].trim() + '\u201d. Use numbers like 1-19, 25, 30-32.' };
    }
  }
  return { idxs:Array.from(set).sort(function(a, b){ return a - b; }) };
}
function compactList(idxs){
  var out = [], i = 0;
  while (i < idxs.length) {
    var j = i; while (j + 1 < idxs.length && idxs[j + 1] === idxs[j] + 1) j++;
    out.push(i === j ? String(idxs[i] + 1) : (idxs[i] + 1) + '\u2013' + (idxs[j] + 1));
    i = j + 1;
  }
  return out.join(', ');
}

function removePages(uids, msg){
  if (!uids.length) return;
  var set = new Set(uids), i = indexOfUid(uids[0]);
  commit(function(){
    S.pages = S.pages.filter(function(p){ return !set.has(p.uid); });
    uids.forEach(function(u){ S.sel.delete(u); });
    if (set.has(S.active)) S.active = S.pages[Math.min(i, S.pages.length - 1)] ? S.pages[Math.min(i, S.pages.length - 1)].uid : null;
  });
  toast(msg || ('Deleted ' + plural(uids.length, 'page') + '.'), 'ok', { label:'Undo', fn:undo });
}

  files = Array.from(files || []);
  if (!files.length) return;
  var ref = (await refAt(at == null ? S.pages.length : at)).size;
  var pages = [], errors = [], nPdf = 0, nImg = 0;
  progress.show('Opening files', 0.05);
  for (var i=0;i<files.length;i++) {

    progress.set('Opening ' + f.name + (files.length > 1 ? ' (' + (i + 1) + ' of ' + files.length + ')' : ''), (i + .5) / files.length);
    try {
      if (isPdfFile(f)) { var s = await loadSource(f); for (var k=0;k<s.count;k++) pages.push(newPage(s.id, k)); nPdf++; }
      else if (isImageFile(f)) { var im = await loadImage(f), sz = imagePageSize(im, S.imgOpts, ref); pages.push(newImagePage(im, sz[0], sz[1], S.imgOpts.margin)); nImg++; }

}

/* ---------- Building the output PDF ----------
   PDF pages: copied as original page objects (no rasterizing, no recompression).
   Image pages: original image data embedded and placed on a vector page.

   Large-file design:
   - Source files are read one at a time. After a file's pages are copied, its parsed
     copy is released, so memory never holds every source at once.
   - The output is written in small pieces straight into a Blob (the browser can keep
     big Blobs on disk), instead of one giant memory buffer. Each object is released as
     soon as it is written. This is what lets 1–2 GB merges finish. */
function readMeta(lib){
  var m = {};
  try { m.title = lib.getTitle(); m.author = lib.getAuthor(); m.subject = lib.getSubject(); m.keywords = lib.getKeywords();
        m.creator = lib.getCreator(); m.producer = lib.getProducer(); m.created = lib.getCreationDate(); } catch(e){}
  return m;
}
async function buildPdf(list){
  progress.show('Preparing PDF', 0.02);
  try {
    var out = await PDFDocument.create({ updateMetadata:false });
    var need = new Map(), totalPdf = 0;
    list.forEach(function(p){ if (p.kind !== 'pdf') return; var m = need.get(p.src) || new Map(); m.set(p.idx, (m.get(p.idx) || 0) + 1); need.set(p.src, m); totalPdf++; });
    var firstPdf = list.find(function(p){ return p.kind === 'pdf'; }), meta = null;
    var copied = new Map(), done = 0, si = 0, nSrc = need.size;
    for (var entry of need) {
      var sid = entry[0], m = entry[1], src = S.sources.get(sid), bytes = null, lib = null;
      si++;
      var tag = nSrc > 1 ? ' (' + si + ' of ' + nSrc + ')' : '';
      progress.set('Reading ' + src.name + ', ' + fmtSize(src.size) + tag, 0.03 + 0.62 * done / Math.max(1, totalPdf)); await tick();
      try { bytes = new Uint8Array(await src.file.arrayBuffer()); }
      catch(e) { throw new Error('"' + src.name + '" couldn\u2019t be read. It may be too large for this browser\u2019s memory, or the file was moved or changed after you added it. Try Chrome or Edge on a computer and close other tabs.'); }
      try { lib = await PDFDocument.load(bytes, { updateMetadata:false, parseSpeed:ParseSpeeds.Fast }); }
      catch(e) { bytes = null; throw new Error('"' + src.name + '" couldn\u2019t be processed' + (/encrypt/i.test(e && e.message) ? ' because it is encrypted.' : '. Its internal structure may be damaged.')); }
      bytes = null;
      if (firstPdf && sid === firstPdf.src) meta = readMeta(lib);
      // One copy pass per source keeps shared fonts/images stored once. Pages used more than once get extra passes.
      for (var round=0;;round++) {
        var idxs = []; m.forEach(function(c, idx){ if (c > round) idxs.push(idx); });
        if (!idxs.length) break;
        for (var a=0;a<idxs.length;a+=200) {
          // Copy in batches so the progress bar moves on documents with thousands of pages.
          var batch = idxs.slice(a, a + 200);
          var pages = await out.copyPages(lib, batch);
          batch.forEach(function(idx, j){ copied.set(sid + '|' + idx + '|' + round, pages[j]); });
          done += batch.length;
          progress.set('Copying pages from ' + src.name + tag + ' \u2014 ' + done + ' of ' + totalPdf, 0.03 + 0.62 * done / Math.max(1, totalPdf)); await tick();
        }
      }
      lib = null; await tick();
    }
    var embedded = new Map(), imgIds = [];
    list.forEach(function(p){ if (p.kind === 'img' && imgIds.indexOf(p.img) < 0) imgIds.push(p.img); });
    for (var j=0;j<imgIds.length;j++) {
      var im = S.images.get(imgIds[j]);
      progress.set('Embedding ' + im.name, 0.65 + 0.05 * j / imgIds.length); await tick();
      try { embedded.set(im.id, im.fmt === 'jpg' ? await out.embedJpg(im.bytes) : await out.embedPng(im.bytes)); }
      catch(e) { throw new Error('The image "' + im.name + '" couldn\u2019t be embedded. Try saving it as JPG or PNG.'); }
    }
    var occ = new Map();
    list.forEach(function(p){
      var pg;
      if (p.kind === 'pdf') {
        var k = p.src + '|' + p.idx, n = occ.get(k) || 0; occ.set(k, n + 1);
        pg = copied.get(k + '|' + n);
        out.addPage(pg);
      } else {
        pg = out.addPage([p.w, p.h]);
        if (p.kind === 'img') drawImageOnPage(pg, embedded.get(p.img), S.images.get(p.img), p);
      }
      if (p.rot) { var a = (pg.getRotation().angle || 0) + p.rot; pg.setRotation(degrees(((a % 360) + 360) % 360)); }
    });
    copied.clear();
    try {
      if (meta) {
        if (meta.title) out.setTitle(meta.title);
        if (meta.author) out.setAuthor(meta.author);
        if (meta.subject) out.setSubject(meta.subject);
        if (meta.keywords) out.setKeywords([meta.keywords]);
        if (meta.creator) out.setCreator(meta.creator);
        if (meta.producer) out.setProducer(meta.producer);
        if (meta.created) out.setCreationDate(meta.created);
      } else { out.setCreationDate(new Date()); }
      out.setModificationDate(new Date());
    } catch(e){}
    if (typeof out.flush === 'function') await out.flush();
    else for (var e of embedded.values()) await e.embed();
    return await writeToBlob(out, function(f){ progress.set('Writing PDF \u2014 ' + Math.round(f * 100) + '%', 0.7 + 0.29 * f); });
  } finally { progress.hide(); }
}
// Streams the finished document into a Blob in ~16 MB pieces with a classic xref table.
async function writeToBlob(doc, onProgress){
  var ctx = doc.context, te = new TextEncoder();
  var parts = [], buf = [], bufSize = 0, offset = 0, CHUNK = 16 * 1048576;
  function put(u8){ buf.push(u8); bufSize += u8.length; offset += u8.length; if (bufSize >= CHUNK) { parts.push(new Blob(buf)); buf = []; bufSize = 0; } }
  function putStr(s){ put(te.encode(s)); }
  put(new Uint8Array([0x25,0x50,0x44,0x46,0x2D,0x31,0x2E,0x37,0x0A,0x25,0x81,0x81,0x81,0x81,0x0A,0x0A])); // %PDF-1.7 + binary marker
  var objs = ctx.enumerateIndirectObjects(), total = objs.length, entries = [];
  for (var i=0;i<total;i++) {
    var ref = objs[i][0], obj = objs[i][1];
    entries.push([ref.objectNumber, offset, ref.generationNumber]);
    putStr(ref.objectNumber + ' ' + ref.generationNumber + ' obj\n');
    var b = new Uint8Array(obj.sizeInBytes()); obj.copyBytesInto(b, 0); put(b);
    putStr('\nendobj\n\n');
    objs[i] = null;
    if (typeof ctx.delete === 'function') ctx.delete(ref);   // free memory as we go
    if (i % 250 === 249) { onProgress(i / total); await tick(); }
  }
  // Cross-reference table, written as runs of consecutive object numbers.
  var xrefAt = offset, pad = function(n, w){ var s = String(n); while (s.length < w) s = '0' + s; return s; };
  entries.sort(function(a, b){ return a[0] - b[0]; });
  entries.unshift([0, 0, 65535, true]);
  var lines = ['xref\n'], k = 0;
  while (k < entries.length) {
    var j = k; while (j + 1 < entries.length && entries[j + 1][0] === entries[j][0] + 1) j++;
    lines.push(entries[k][0] + ' ' + (j - k + 1) + '\n');
    for (var q=k;q<=j;q++) {
      var en = entries[q];
      lines.push(en[3] ? '0000000000 65535 f\r\n' : pad(en[1], 10) + ' ' + pad(en[2], 5) + ' n\r\n');
      if (lines.length >= 50000) { putStr(lines.join('')); lines = []; }
    }
    k = j + 1;
  }
  putStr(lines.join(''));
  var ti = ctx.trailerInfo, t = { Size:ctx.largestObjectNumber + 1, Root:ti.Root };
  if (ti.Info) t.Info = ti.Info;
  if (ti.ID) t.ID = ti.ID;
  putStr('trailer\n' + ctx.obj(t).toString() + '\nstartxref\n' + xrefAt + '\n%%EOF\n');
  if (buf.length) parts.push(new Blob(buf));
  onProgress(1);
  return new Blob(parts, { type:'application/pdf' });
}
// Places the image at full resolution, fitted inside the page. EXIF rotation is applied as a
// drawing transform, so JPEG data stays byte-for-byte unchanged.
function drawImageOnPage(pg, emb, im, p){
  var b = fitBox(p.w, p.h, im.w, im.h, p.m), o = im.orient;
  if (o === 6) pg.drawImage(emb, { x:b.x, y:b.y + b.h, width:b.h, height:b.w, rotate:degrees(-90) });
  else if (o === 8) pg.drawImage(emb, { x:b.x + b.w, y:b.y, width:b.h, height:b.w, rotate:degrees(90) });
  else if (o === 3) pg.drawImage(emb, { x:b.x + b.w, y:b.y + b.h, width:b.w, height:b.h, rotate:degrees(180) });
  else pg.drawImage(emb, { x:b.x, y:b.y, width:b.w, height:b.h });
}
async function saveFile(blob, filename){
  var dl = await downloadsCap;
  if (dl) {
    try { await dl.save({ filename:filename, data:blob }); toast('Saved ' + filename + ' (' + fmtSize(blob.size) + ').', 'ok'); return; }
    catch(e) {
      var c = e && e.code;
      if (c === 'declined') { toast('Download cancelled.', 'info'); return; }

      if (c === 'too_large') { toast('This file is too large to save in this app. Open the page in a desktop browser to save it.', 'error'); return; }
    }
  }
  var url = URL.createObjectURL(blob), a = document.createElement('a');
  a.href = url; a.download = filename; document.body.appendChild(a); a.click(); a.remove();
  setTimeout(function(){ URL.revokeObjectURL(url); }, 120000);

}

  }
  await saveFile(blob, name);
}

function dropIndexAt(x, y){
  var best = null, bd = Infinity;
  for (var i=0;i<S.pages.length;i++) {
    var c = cards.get(S.pages[i].uid); if (!c) continue;
    var r = c.getBoundingClientRect();
    var dx = Math.max(r.left - x, 0, x - r.right), dy = Math.max(r.top - y, 0, y - r.bottom);
    var d = dx * dx + dy * dy * 4;
    if (d < bd) { bd = d; best = { i:i, r:r }; }
    if (d === 0) break;
  }
  if (!best) return null;
  var after = x > best.r.left + best.r.width / 2;
  return { at:best.i + (after ? 1 : 0), r:best.r, after:after };
}

    await insertFiles(files, 0);
    return;
  }
  await insertFiles(files, at == null ? S.pages.length : at);
